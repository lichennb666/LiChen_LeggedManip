cmake_minimum_required(VERSION 3.8)
project(go2_piper_gazebo_plugins)
find_package(ament_cmake REQUIRED)
find_package(gazebo REQUIRED)
find_package(gazebo_ros REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_srvs REQUIRED)
add_library(go2_piper_grasp_plugin SHARED src/grasp_plugin.cpp)
add_library(go2_piper_base_stabilizer SHARED src/base_stabilizer.cpp)
target_include_directories(go2_piper_grasp_plugin PRIVATE ${GAZEBO_INCLUDE_DIRS})
target_include_directories(go2_piper_base_stabilizer PRIVATE ${GAZEBO_INCLUDE_DIRS})
target_link_libraries(go2_piper_grasp_plugin ${GAZEBO_LIBRARIES})
target_link_libraries(go2_piper_base_stabilizer ${GAZEBO_LIBRARIES})
ament_target_dependencies(go2_piper_base_stabilizer gazebo_ros rclcpp geometry_msgs)
ament_target_dependencies(go2_piper_grasp_plugin gazebo_ros rclcpp std_srvs)
install(TARGETS go2_piper_grasp_plugin go2_piper_base_stabilizer LIBRARY DESTINATION lib)
ament_package()
